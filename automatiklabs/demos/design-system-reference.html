CODE BLOCKS

Terminal-like code containers. bg-inset background, 2px border, header bar with language label and copy action. Syntax colors: keyword=blue, string=cyan, function=amber, comment=text-3, number=green.

typescript copiar 
// AutomatikClub AI Pipeline
import { Anthropic } from "@anthropic-ai/sdk";

const client = new Anthropic();
const maxTokens = 1024;

async function analyzeCode(input: string) {
  const msg = await client.messages.create({
    model: "claude-sonnet-4-20250514",
    max_tokens: maxTokens,
    messages: [{
      role: "user",
      content: input
    }]
  });
  return msg.content[0].text;
}
